cmake_minimum_required(VERSION 3.16)
project(AirportOps LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(airportops STATIC src/airportops.cpp)
target_include_directories(airportops PUBLIC include)
add_executable(airportops_cli src/main.cpp)
target_link_libraries(airportops_cli PRIVATE airportops)
enable_testing()
add_executable(airportops_tests tests/test_engine.cpp)
target_link_libraries(airportops_tests PRIVATE airportops)
add_test(NAME engine COMMAND airportops_tests)
